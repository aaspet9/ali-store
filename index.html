async function verifySingleImage(index) { if (!proofFiles[index]) return; const loader = document.getElementById('proofLoader'); const verifyBtn = document.getElementById(`verifyBtn${index}`); const badge = document.getElementById(`badge${index}`); loader.style.display = 'block'; loader.innerText = `Scanning Link for Screenshot #${index}... 🔍⏳`; verifyBtn.disabled = true; // 1. پروڈکٹ کے اصل لنک کا کوڈ ایکسٹریکٹ کریں let targetLinkKey = extractUrlKey(selectedProdLink).toLowerCase().replace(/[^a-z0-9]/g, ''); try { const ocrResult = await Tesseract.recognize(proofFiles[index], 'eng'); const scannedText = ocrResult.data.text.toLowerCase().replace(/[^a-z0-9]/g, ''); // 2. مین فوکس: کیا تصویر کے اندر افیلی ایٹ لنک موجود ہے؟ let isLinkMatch = (targetLinkKey.length > 3 && scannedText.includes(targetLinkKey)); if (!isLinkMatch) { // اگر لنک بالکل اسکین نہ ہو سکے تو ہی ٹائٹل کے اہم لفظ کو سیکنڈری چیک کریں const targetKeywords = getKeywordsFromProd(selectedProdTitle, ""); let isTitleMatch = targetKeywords.some(kw => scannedText.includes(kw)); if (!isTitleMatch) { loader.style.display = "none"; badge.style.display = 'inline-block'; badge.className = 'status-badge not-verified'; badge.innerText = "Error ❌"; verifyBtn.disabled = false; verifiedStates[index] = false; checkAllVerificationStatuses(); alert(`❌ اسکرین شاٹ #${index} میں افیلی ایٹ لنک موجود نہیں ہے۔ برائے مہربانی صحیح لنک والا اسکرین شاٹ اپلوڈ کریں!`); return; } } // 3. لنک کی تصدیق ہونے پر امیج اپلوڈ کریں loader.innerText = `Uploading Verified Screenshot #${index}... 🚀`; const formData = new FormData(); formData.append("image", proofFiles[index]); const res = await fetch(`https://api.imgbb.com/1/upload?key=${IMGBB_API_KEY}`, { method: "POST", body: formData }); const imgData = await res.json(); if (imgData.success) { uploadedUrls[index] = imgData.data.url; verifiedStates[index] = true; badge.style.display = 'inline-block'; badge.className = 'status-badge verified'; badge.innerText = "Verified ✅"; verifyBtn.disabled = true; } else { verifyBtn.disabled = false; } } catch(e) { verifyBtn.disabled = false; alert("اسکیننگ کے دوران مسئلہ آیا، دوبارہ کوشش کریں۔"); } loader.style.display = 'none'; checkAllVerificationStatuses(); }
